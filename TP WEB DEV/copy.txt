let heures = document.getElementsByName("hour");
            let somme = 0;
            
            for(i=0;i<heures.length;i++){
                somme = somme + Number(heures[i].value);
            }
            
            if(somme > 24){
                let dayAndHour = somme / 24;
                day.value = Math.floor(somme / 24);
                totalH.value = dayAndHour - day.value;
            }
            else{
                totalH.value = somme;
                day.value = 0;
            }

            let minutes = document.getElementsByName("min");
            let sum = 0;

            for(i=0;i<minutes.length;i++){
                sum = sum + Number(minutes[i].value);
            }
            totalM.value = sum ;

            let secondes = document.getElementsByName("sec");
            let summ = 0;

            for(i=0;i<secondes.length;i++){
                summ = summ + Number(secondes[i].value);
            }
            totalS.value = summ;
